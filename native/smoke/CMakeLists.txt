cmake_minimum_required(VERSION 3.22)
project(encoder_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(MINGW)
    add_compile_options(-include "${CMAKE_CURRENT_SOURCE_DIR}/mingw_compat.h")
endif()
add_subdirectory(../../third_party/whisper.cpp whisper)
add_executable(encoder-smoke smoke.cpp)
target_link_libraries(encoder-smoke PRIVATE whisper)
