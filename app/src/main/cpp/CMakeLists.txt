cmake_minimum_required(VERSION 3.22.1)
project(localvoices LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized inference in every APK" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    # SSE4.2 is part of Android's x86_64 ABI baseline; AVX is deliberately optional.
    set(GGML_SSE42 ON CACHE BOOL "" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/whisper.cpp whisper)
add_library(voices SHARED bridge.cpp)
target_link_libraries(voices PRIVATE whisper android log)
target_link_options(voices PRIVATE "-Wl,-z,max-page-size=16384")
target_link_options(voices PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
