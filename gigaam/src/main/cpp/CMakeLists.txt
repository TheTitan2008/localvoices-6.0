cmake_minimum_required(VERSION 3.22.1)
project(localvoices_gigaam LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_VULKAN OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/transcribe.cpp transcribe)
set_target_properties(transcribe PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(transcribe PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/transcribe.cpp/include)
add_library(gigaam SHARED gigaam_bridge.cpp)
target_include_directories(gigaam PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/transcribe.cpp/include)
target_link_libraries(gigaam PRIVATE transcribe android log)
target_link_options(gigaam PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
